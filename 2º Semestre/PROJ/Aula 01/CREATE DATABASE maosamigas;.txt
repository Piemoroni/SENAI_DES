DROP DATABASE IF EXISTS lojao;
CREATE DATABASE maosamigas;

USE maosamigas;

CREATE TABLE usuarios (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(150) NOT NULL,
  email_institucional VARCHAR(150) NOT NULL,
  senha VARCHAR(150) NOT NULL
);

CREATE TABLE alunos (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  usuario_id INTEGER,
  turma VARCHAR(50) NOT NULL,
  unidade_escolar VARCHAR(150) NOT NULL,
  FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE professores (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  usuario_id INTEGER,
  disciplinas VARCHAR(150) NOT NULL,
  turma VARCHAR(50) NOT NULL,
  unidade_escolar VARCHAR(150) NOT NULL,
  FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE autoridades_escolares (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  usuario_id INTEGER,
  nivel_escolaridade VARCHAR(50) NOT NULL,
  unidade_escolar VARCHAR(150) NOT NULL,
  FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE familia (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  usuario_id INTEGER,
  aluno_id INTEGER,
  FOREIGN KEY (usuario_id) REFERENCES usuarios(id),
  FOREIGN KEY (aluno_id) REFERENCES alunos(id)
);

CREATE TABLE denuncias (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  aluno_id INTEGER,
  descricao VARCHAR(150) NOT NULL,
  data_ocorrido DATETIME NOT NULL,
  responsavel_ocorrido VARCHAR(150) NOT NULL,
  FOREIGN KEY (aluno_id) REFERENCES alunos(id)
);

CREATE TABLE painel_acompanhamento_alunos (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  aluno_id INTEGER,
  responsavel_id INTEGER,
  descricao_postagem VARCHAR(150) NOT NULL,
  data_acao DATETIME NOT NULL,
  FOREIGN KEY (aluno_id) REFERENCES alunos(id),
  FOREIGN KEY (responsavel_id) REFERENCES familia(id)
);

CREATE TABLE painel_acompanhamento_autoridades (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  autoridade_id INTEGER,
  descricao_postagem VARCHAR(150) NOT NULL,
  data_acao DATETIME NOT NULL,
  FOREIGN KEY (autoridade_id) REFERENCES autoridades_escolares(id)
);

CREATE TABLE central_opinioes (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  aluno_id INTEGER,
  descricao_opiniao VARCHAR(150) NOT NULL,
  FOREIGN KEY (aluno_id) REFERENCES alunos(id)
);

CREATE TABLE ia_psicologa (
  id INTEGER AUTO_INCREMENT PRIMARY KEY,
  aluno_id INTEGER,
  descricao_conversa VARCHAR(150) NOT NULL,
  FOREIGN KEY (aluno_id) REFERENCES alunos(id)
);
